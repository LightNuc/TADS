	ESCREVA "Objeto ", i
	ESCREVA "Quantidade vendida: ", quantidade[i]

	ESCREVA "Valor total: ", total[i]

	SE quantidade[i] > maiorQuantidade
	ENTAO INICIO
		maiorQuantidade <- quantidade[i]
		posição <- i
	FIM

maiorQuantidade, " unidades"
